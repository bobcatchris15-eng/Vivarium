using System;
using System.Collections.Generic;
using Vivarium.Sim.Core;

namespace Vivarium.Sim.Repr.Plant.Forms;

/// <summary>
/// One generator class per growth form. Concrete classes only describe the topology of their form (Build); this base
/// calibrates it: per-individual size, height and crown radius via dry runs, and a
/// leaf area scale that hits the trait LeafAreaIndex. Structure is addressed by hashed identity, so the calibration
/// scales change sizes and thresholds without reshuffling the plant.
/// </summary>
public abstract class FormGenerator : PlanGenerator
{
    public string Species { get; }
    public FormTraits Traits { get; }
    public override string SpeciesId => Species;

    protected FormGenerator(string species, FormTraits traits)
    {
        Species = species;
        Traits = traits;
    }

    internal abstract void Build(FormCtx c);

    /// <summary>Ground area the LAI is measured against (m2).</summary>
    public double CrownArea => Math.PI * Traits.CrownRadiusM * Traits.CrownRadiusM;

    public double TargetLeafArea => Traits.LeafAreaIndex * CrownArea;

    /// <summary>Diagnostics of the last Design: leaf area at AreaScale 1 and the calibration chosen.</summary>
    public double BaseLeafArea;
    public (double LenScale, double Density, double AreaScale) LastCalibration;

    protected override void Design(PlanBuilder b)
    {
        var t = Traits;
        double hk = 1.0;
        double ind = 1.0;
        for (int it = 0; it < 3; it++)
        {
            var d = new PlanBuilder(b.WorldSeed, b.PlantId);
            var c = new FormCtx(d, t, Species) { LenScale = hk };
            ind = c.IndScale;
            Build(c);
            var m = d.Measure();
            double rH = t.HeightM * ind / Math.Max(0.02, m.Height);
            double rC = t.CrownRadiusM * ind / Math.Max(0.03, m.Crown);
            double r = Math.Pow(rH, 0.65) * Math.Pow(rC, 0.35);
            if (Math.Abs(r - 1) < 0.04) break;
            hk *= Math.Clamp(r, 0.55, 1.8);
        }
        // A render budget must not suppress organs in the authoritative plant plan.
        // Reduce detail in the compiled representation instead of thinning the crown.
        const double density = 1.0;
        double areaScale = 1.0;
        {
            var d = new PlanBuilder(b.WorldSeed, b.PlantId);
            Build(new FormCtx(d, t, Species) { LenScale = hk, Density = density });
            var m = d.Measure();
            if (m.LeafArea > 1e-6) areaScale = Math.Clamp(TargetLeafArea / m.LeafArea, 0.35, 3.0);
            BaseLeafArea = m.LeafArea;
        }
        LastCalibration = (hk, density, areaScale);
        Build(new FormCtx(b, t, Species) { LenScale = hk, Density = density, AreaScale = areaScale });
    }

    // ---- helpers shared by concrete generators

    internal static void Finalise(FormCtx c, Ax root, Vec3 startDir)
    {
        c.ComputeRadii(root);
        c.Emit(root, -1, startDir, 0.0);
    }
}

/// <summary>Multi-stemmed shrub: a short stool, arching canes that sag and lift at the tips, orders of side branches.</summary>
public sealed class MultiStemShrubGenerator : FormGenerator
{
    public MultiStemShrubGenerator(string species, FormTraits traits) : base(species, traits) { }

    internal override void Build(FormCtx c)
    {
        var T = c.T;
        c.MakeShapes();
        if (T.Inflorescence != InflorescenceType.None) c.FlowerShapeId = c.AddFlowerShape(T.AsFruit);
        ulong rid = c.RootId;
        int n = T.StemMin + (int)(FormCtx.U(rid, 21) * (T.StemMax - T.StemMin + 1));
        n = Math.Clamp(n, T.StemMin, T.StemMax);
        double H = T.HeightM * c.IndScale * c.LenScale;
        c.MainLen = H * 1.3;
        var stool = c.NewAxis(FormCtx.Child(rid, 1), 0, 0.07, 0.07, Tissue.Wood, 0.10, null, 0);
        stool.Root = true; stool.CanBranch = false; stool.SagMul = 0;
        stool.Spec.Wander = 0.02;
        double rot = FormCtx.U(rid, 22) * Math.Tau;
        for (int i = 0; i < n; i++)
        {
            ulong id = FormCtx.Child(rid, 10 + i);
            double az = rot + i * Math.Tau / n + FormCtx.Ur(id, 1, -0.35, 0.35);
            double ab = FormCtx.U(id, 2) < 0.1 ? 0.45 : 1.0;
            double len = H * FormCtx.Ur(id, 3, 1.0, 1.4) * ab;
            double el = FormCtx.Rad(T.LeaderElevDeg) + FormCtx.Ur(id, 4, -0.22, 0.22);
            var cane = c.NewAxis(id, 1, len, T.InternodeM * 1.2, T.BarkTissue ? Tissue.Bark : Tissue.Wood, 0.36, stool, 0);
            cane.LeafBearing = T.LeafMinOrder <= 1;
            cane.LeafLoad = cane.LeafBearing ? 0.5 : 0.25;
            cane.BranchStart = 0.16;
            cane.StartDir = FormCtx.DirFrom(el, az);
            cane.AttachT = 0.3 + 0.6 * FormCtx.U(id, 6);
            cane.Lag = FormCtx.Ur(id, 5, 0.0, 0.26);
            c.Populate(cane, T.ThornsPerM);
            stool.Sites.Add(new Site { Seg = 0, T = cane.AttachT, Az = az, S = 0.5, Kind = SiteKind.Branch, Child = cane });
        }
        foreach (var a in c.All) a.FlowerHost = a.Order >= Math.Max(1, T.BranchOrders - 1) && a.NSeg >= 2 && a.Order > 0;
        c.PlaceInflorescences();
        Finalise(c, stool, Vec3.Up);
    }
}

/// <summary>Single-trunk tree: leader with a clear bole, paired primary branches under a crown envelope, sub-branches and twigs.</summary>
public sealed class SingleTrunkTreeGenerator : FormGenerator
{
    public SingleTrunkTreeGenerator(string species, FormTraits traits) : base(species, traits) { }

    internal override void Build(FormCtx c)
    {
        var T = c.T;
        c.MakeShapes();
        if (T.Inflorescence != InflorescenceType.None) c.FlowerShapeId = c.AddFlowerShape(T.AsFruit);
        ulong rid = c.RootId;
        double H = T.HeightM * c.IndScale * c.LenScale;
        double R = T.CrownRadiusM * c.IndScale * c.LenScale;
        c.MainLen = H;
        double bole = 0.26 + 0.14 * FormCtx.U(rid, 3) + 0.05 * (T.ApicalDominance - 0.5);
        double trunkLen = H * 0.93;
        double lean = FormCtx.Ur(rid, 4, 0, 0.16);
        var trunk = c.NewAxis(FormCtx.Child(rid, 1), 0, trunkLen, trunkLen / 13.0, T.BarkTissue ? Tissue.Bark : Tissue.Wood, 0.46, null, 0);
        trunk.Root = true;
        trunk.BranchStart = bole;
        trunk.BranchesPerNode = 2;
        trunk.BranchProbMul = 2.0;
        trunk.LeafBearing = false;
        trunk.SagMul = 0.4;
        trunk.Spec.Wander = FormCtx.Rad(3.5);
        trunk.StartDir = FormCtx.DirFrom(Math.PI / 2 - lean, FormCtx.U(rid, 5) * Math.Tau);
        double apex = 1.0 - bole;
        trunk.LateralLength = s =>
        {
            double u = Math.Clamp((1.0 - s) / apex, 0, 1);
            return R * (0.28 + 0.95 * Math.Pow(u, 0.65)) * (0.8 + 0.2 * (1 - T.ApicalDominance));
        };
        c.Populate(trunk, T.ThornsPerM);
        foreach (var a in c.All) a.FlowerHost = a.Order >= Math.Max(1, T.BranchOrders - 1) && a.NSeg >= 2;
        c.PlaceInflorescences();
        Finalise(c, trunk, trunk.StartDir!.Value);
    }
}

/// <summary>Fern rosette: a short crown, fronds on a 137.5 degree spiral, each an arching, twisting rachis with pinna pairs; croziers when young.</summary>
public sealed class FernRosetteGenerator : FormGenerator
{
    public FernRosetteGenerator(string species, FormTraits traits) : base(species, traits) { }

    internal override void Build(FormCtx c)
    {
        var T = c.T;
        ulong rid = c.RootId;
        double jit = (FormCtx.U(rid, 11) * 2 - 1) * T.ColourVariance;
        var shapes = new List<ushort>();
        for (int young = 0; young < 2; young++)
            for (int v = 0; v < 3; v++)
                shapes.Add(c.B.AddShape(FormShapes.Leaflet(T, $"{c.Species}_pinna{v}{(young == 1 ? "y" : "")}", v, young == 1, jit, !T.StrapFronds)));
        c.LeafletShapes = shapes.ToArray();
        if (T.StrapFronds)
        {
            var bl = new List<ushort>();
            for (int v = 0; v < 6; v++)
                bl.Add(c.B.AddShape(FormShapes.Blade(T, $"{c.Species}_strap{v}", T.BladeCurl * (0.55 + 0.16 * v), 0.15 + 0.03 * v, (v - 2.5) * 0.08, v >= 5, jit)));
            c.BladeShapes = bl.ToArray();
        }
        if (T.Inflorescence != InflorescenceType.None) c.FlowerShapeId = c.AddFlowerShape(T.AsFruit);

        int n = Math.Clamp(T.FrondMin + (int)(FormCtx.U(rid, 21) * (T.FrondMax - T.FrondMin + 1)), T.FrondMin, T.FrondMax);
        double FL = T.FrondLengthM * c.IndScale * c.LenScale;
        c.FrondLength = FL;
        c.MainLen = FL;
        var crown = c.NewAxis(FormCtx.Child(rid, 1), 0, 0.03 + 0.03 * FL, 0.06, Tissue.Bark, 0.08, null, 0);
        crown.Root = true; crown.CanBranch = false; crown.SagMul = 0; crown.Spec.Wander = 0.01;
        double rot = FormCtx.U(rid, 22) * Math.Tau;
        var frs = new List<Ax>();
        for (int i = 0; i < n; i++)
        {
            ulong id = FormCtx.Child(rid, 10 + i);
            double rank = n == 1 ? 0.5 : i / (double)(n - 1);       // 0 = oldest (outer), 1 = youngest (centre)
            double az = rot + i * FormCtx.Rad(137.5) + FormCtx.Ur(id, 1, -0.2, 0.2);
            double len = FL * FormCtx.Ur(id, 2, 0.78, 1.12) * (1.0 - 0.28 * rank * rank);
            double el = FormCtx.Rad(MathD.Lerp(42, 80, rank) + FormCtx.Ur(id, 3, -8, 8));
            double internode = T.StrapFronds ? len : len / Math.Max(6, T.PinnaPairs * 1.6);
            var fr = c.NewAxis(id, 1, len, internode, Tissue.Green, 0.5, crown, 0);
            fr.Spec.SetPoint = FormCtx.Rad(-12 - 14 * (1 - rank));
            fr.SagMul = 0.25;
            fr.Spec.SetRate = T.GravRate;
            fr.Spec.Twist = FormCtx.Ur(id, 4, -0.5, 0.5);
            fr.Spec.Wander = FormCtx.Rad(T.WanderDeg);
            fr.Coil = 3.4;
            fr.CanBranch = false;
            fr.StartDir = FormCtx.DirFrom(el, az);
            fr.AttachT = 0.4 + 0.55 * FormCtx.U(id, 5);
            fr.Lag = 0.02 + 0.5 * rank * FormCtx.Ur(id, 6, 0.7, 1.0);
            frs.Add(fr);
            if (!T.StrapFronds)
            {
                for (int k = 1; k < fr.NSeg; k++)
                    fr.Sites.Add(new Site { Seg = k, T = 0.5, Az = az, S = fr.CumFrac[k] - 0.5 * fr.SegLen[k] / fr.Length, Kind = SiteKind.PinnaPair });
                fr.Sites.Add(new Site { Seg = fr.NSeg - 1, T = 0.985, Az = az, S = 1.0, Kind = SiteKind.PinnaPair, A = 1 });
            }
            crown.Sites.Add(new Site { Seg = 0, T = fr.AttachT, Az = az, S = 0.5, Kind = SiteKind.Branch, Child = fr });
        }
        Finalise(c, crown, Vec3.Up);
        if (T.StrapFronds)
        {
            foreach (var fr in frs)
            {
                int g = fr.Gen[^1];
                var tn = c.B.EndDir(g);
                var (em, du) = c.B.Timing(g);
                ulong id = FormCtx.Child(fr.Id, 77);
                double len = T.BladeLengthM * c.IndScale * c.LenScale * FormCtx.Ur(id, 1, 0.8, 1.15);
                double w = T.BladeWidthM * c.AreaScale * FormCtx.Ur(id, 2, 0.85, 1.15);
                int sv = Math.Clamp((int)(FormCtx.U(id, 3) * 5), 0, 4);
                c.EmitBlade(g, 0.98, Math.Atan2(tn.Z, tn.X), Math.Clamp(Math.Asin(Math.Clamp(tn.Y, -1, 1)), 0.1, 1.2) * 0.9, len, w,
                    c.BladeShapes[sv], 2, em + du + 0.004, 0.32, FormCtx.Ur(id, 4, -0.1, 0.1));
            }
        }
    }
}

/// <summary>Leafy upright forb: flexuous stems with nutation, phyllotactic leaves graded in size, side shoots high up, terminal inflorescences.</summary>
public sealed class ForbGenerator : FormGenerator
{
    public ForbGenerator(string species, FormTraits traits) : base(species, traits) { }

    internal override void Build(FormCtx c)
    {
        var T = c.T;
        c.MakeShapes();
        if (T.Inflorescence != InflorescenceType.None) c.FlowerShapeId = c.AddFlowerShape(T.AsFruit);
        ulong rid = c.RootId;
        int n = Math.Clamp(T.StemMin + (int)(FormCtx.U(rid, 21) * (T.StemMax - T.StemMin + 1)), T.StemMin, T.StemMax);
        double H = T.HeightM * c.IndScale * c.LenScale;
        c.MainLen = H;
        var crown = c.NewAxis(FormCtx.Child(rid, 1), 0, 0.02, 0.02, Tissue.Green, 0.08, null, 0);
        crown.Root = true; crown.CanBranch = false; crown.SagMul = 0; crown.Spec.Wander = 0.01;
        double rot = FormCtx.U(rid, 22) * Math.Tau;
        for (int i = 0; i < n; i++)
        {
            ulong id = FormCtx.Child(rid, 10 + i);
            double az = rot + i * Math.Tau / n + FormCtx.Ur(id, 1, -0.4, 0.4);
            double len = H * FormCtx.Ur(id, 3, 0.85, 1.12) * (i == 0 ? 1.0 : FormCtx.Ur(id, 7, 0.75, 1.0));
            double el = FormCtx.Rad(FormCtx.Ur(id, 4, 50, 82));
            var st = c.NewAxis(id, 1, len, T.InternodeM, Tissue.Green, 0.5, crown, 0);
            st.LeafBearing = T.LeafMinOrder <= 1;
            st.LeafLoad = 0.6;
            st.BranchStart = 0.38;
            st.Spec.Twist = FormCtx.Ur(id, 8, -0.6, 0.6);
            st.StartDir = FormCtx.DirFrom(el, az);
            st.AttachT = 0.4 + 0.5 * FormCtx.U(id, 6);
            st.Lag = FormCtx.Ur(id, 5, 0.0, 0.2);
            c.Populate(st, T.ThornsPerM);
            crown.Sites.Add(new Site { Seg = 0, T = st.AttachT, Az = az, S = 0.5, Kind = SiteKind.Branch, Child = st });
        }
        foreach (var a in c.All) a.FlowerHost = a.Order >= 1 && a.NSeg >= 2;
        c.PlaceInflorescences();
        Finalise(c, crown, Vec3.Up);
    }
}

/// <summary>Graminoid tussock: many arched, twisted, partly folded blades in a spiral, plus culms carrying seed heads.</summary>
public sealed class GraminoidGenerator : FormGenerator
{
    public GraminoidGenerator(string species, FormTraits traits) : base(species, traits) { }

    internal override void Build(FormCtx c)
    {
        var T = c.T;
        ulong rid = c.RootId;
        double jit = (FormCtx.U(rid, 11) * 2 - 1) * T.ColourVariance;
        var bl = new List<ushort>();
        for (int v = 0; v < 6; v++)
        {
            double f = 0.45 + 0.22 * v;
            bl.Add(c.B.AddShape(FormShapes.Blade(T, $"{c.Species}_blade{v}", T.BladeCurl * f, v % 3 == 0 ? 0.45 : 0.05 + 0.05 * v, (v - 2.5) * 0.15, v == 0, jit)));
        }
        c.BladeShapes = bl.ToArray();
        if (T.Inflorescence != InflorescenceType.None) c.FlowerShapeId = c.AddFlowerShape(T.AsFruit);

        double H = T.HeightM * c.IndScale * c.LenScale;
        c.MainLen = H;
        var crown = c.NewAxis(FormCtx.Child(rid, 1), 0, 0.025, 0.025, Tissue.Green, 0.06, null, 0);
        crown.Root = true; crown.CanBranch = false; crown.SagMul = 0; crown.Spec.Wander = 0.01;
        crown.BaseRadiusMul = 1.0;
        int nb = T.BladeMin + (int)(FormCtx.U(rid, 21) * (T.BladeMax - T.BladeMin + 1));
        nb = Math.Max(4, (int)(nb * Math.Max(0.35, c.Density)));
        int nc = Math.Clamp(T.CulmMin + (int)(FormCtx.U(rid, 23) * (T.CulmMax - T.CulmMin + 1)), T.CulmMin, T.CulmMax);
        if (T.Inflorescence == InflorescenceType.None) nc = 0;
        double rot = FormCtx.U(rid, 22) * Math.Tau;
        for (int i = 0; i < nc; i++)
        {
            ulong id = FormCtx.Child(rid, 70 + i);
            double az = rot + i * Math.Tau / Math.Max(1, nc) + FormCtx.Ur(id, 1, -0.5, 0.5);
            double len = H * FormCtx.Ur(id, 3, 0.8, 1.0);
            double el = FormCtx.Rad(FormCtx.Ur(id, 4, 48, 70));
            var cu = c.NewAxis(id, 1, len, Math.Max(0.05, len / 6.0), Tissue.Green, 0.5, crown, 0);
            cu.SagMul = 0.10;
            cu.Spec.SetPoint = FormCtx.Rad(86);
            cu.Spec.SetRate = 0.55;
            cu.Spec.Wander = FormCtx.Rad(6);
            cu.CanBranch = false;
            cu.FlowerHost = true;
            cu.StartDir = FormCtx.DirFrom(el, az);
            cu.AttachT = 0.4 + 0.5 * FormCtx.U(id, 6);
            cu.Lag = FormCtx.Ur(id, 5, 0.1, 0.35);
            crown.Sites.Add(new Site { Seg = 0, T = cu.AttachT, Az = az, S = 0.5, Kind = SiteKind.Branch, Child = cu });
        }
        // every culm carries a head
        foreach (var a in c.All.ToArray())
            if (a.Order == 1 && a.FlowerHost)
                a.Sites.Add(new Site { Seg = a.NSeg - 1, T = 0.92, Az = 0, S = 1, Kind = SiteKind.Infl, A = 1 });
        c.ComputeRadii(crown);
        c.Emit(crown, -1, Vec3.Up, 0.0);
        int g0 = crown.Gen[0];
        for (int i = 0; i < nb; i++)
        {
            ulong id = FormCtx.Child(rid, 200 + i);
            double rank = FormCtx.U(id, 1);                      // 0 inner .. 1 outer
            double az = rot + i * FormCtx.Rad(137.508) + FormCtx.Ur(id, 2, -0.15, 0.15);
            double len = T.BladeLengthM * c.IndScale * c.LenScale * FormCtx.Ur(id, 3, 0.5, 1.12);
            double el = FormCtx.Rad(MathD.Lerp(82, 34, Math.Pow(rank, 0.8)) + FormCtx.Ur(id, 4, -9, 9));
            double w = T.BladeWidthM * c.AreaScale * FormCtx.Ur(id, 5, 0.8, 1.2) * (0.6 + 0.4 * Math.Min(1, len / T.BladeLengthM));
            int sv = Math.Clamp((int)(rank * 3.2) + (FormCtx.U(id, 6) < 0.5 ? 0 : 1) + (rank > 0.7 ? 1 : 0), 0, 5);
            double emerge = 0.03 + 0.62 * Math.Pow(FormCtx.U(id, 7), 1.2);
            c.EmitBlade(g0, FormCtx.Ur(id, 8, 0.1, 0.9), az, el, len, w, c.BladeShapes[sv], 1, emerge, 0.30, FormCtx.Ur(id, 9, -0.15, 0.15));
        }
    }
}
