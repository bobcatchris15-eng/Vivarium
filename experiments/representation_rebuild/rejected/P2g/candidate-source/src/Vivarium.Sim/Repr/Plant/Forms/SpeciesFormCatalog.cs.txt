namespace Vivarium.Sim.Repr.Plant.Forms;

/// <summary>
/// Maps a species (its visual Shape string plus size and colour values read by the game) to a growth form and a trait
/// preset. The sim side never reads files: the caller passes the values in. Per-species variation (leaf size and
/// aspect, branch angle, internode, sag, wander, phyllotaxy tie-break, colour tint) is derived deterministically from
/// the species id hash inside the preset's plausible ranges.
///
/// Shape -> form mapping (Shape = visual.shape of game/content/flora/*.json). Implemented forms only; anything else
/// returns false and the legacy mesh stays.
///   MultiStemShrub : shrub_glowbriar shrub_shadebell shrub_embercrown shrub_lanternbrush greywatch hookthicket_brake
///                    miremyrtle nutglade silverfelt thornmantle microcanopy_splitcrown
///   SingleTrunkTree: banksorrow fenforge hollowcrown_tree mirespire moonbark shiverleaf witchberry microcanopy_lacemantle
///   FernRosette    : fern (moonfern, veilfern) sceptrefern swanfrond emberbracken mirrorstrap_fern
///   Forb           : blackneedle_thistle hallowsting heddlelace widowspire marrowbloom_stem fenfoam butterglint
///   Graminoid      : tussock pampas quillsedge cloudtuft bannerreed coalspike_cattail veilblade_curtain ashwire
///                    duskbristle reed (glassrush lanternsedge ringreed) sootband ribbonweed
/// Not implemented (legacy stays): carpet, cushion, crust, foliose, fruticose, capitula and the other mosses/lichens,
/// all fungi (mushroom_cluster, bracket, ...), creeping mats (creeper, roundleaf, trifoliate, iceplant, sundew_mat,
/// pairedleaf, floatleaf, glasslily_pad, milfoil, submerged_whorl), climbers (vine_*, trumpetbind, bitterbine,
/// shadewrap), rosettes (herb, succulent, tankbloom_bromeliad, pitcher_rosette, snaptrap_rosette, starbulb_umbel,
/// burlmantle, cragset, ...), kinkcane_brake, microcanopy_cragparasol/rainshield/vaultfern, paddlerib, barbarrow,
/// burrflag, deepquill, mirepocket, starveil, mistviolet, marshkettle, hushclover, etc.
/// </summary>
public static class SpeciesFormCatalog

    /// <summary>Shape strings handled by this catalog.</summary>
    public static IReadOnlyCollection<string> MappedShapes => FormOfShape.Keys;

    public static bool TryGetForm(string shape, out GrowthForm form) => FormOfShape.TryGetValue(shape, out form);

    private static ulong IdHash(string id)
    {
        ulong h = 1469598103934665603UL;
        foreach (char ch in id) h = (h ^ ch) * 1099511628211UL;
        return h;
    }

    /// <param name="colour">json visual.color (sRGB 0..1)</param>
    /// <param name="colour2">json visual.color2 (sRGB 0..1)</param>
    /// <param name="glow">json bioluminescenceColor, or null / black if the species does not glow</param>
    public static bool TryGetTraits(string speciesId, string shape, double heightM, double radiusM, double[] colour, double[] colour2,
        double[]? glow, out FormTraits traits)
    {
        traits = null!;
        if (!FormOfShape.TryGetValue(shape, out var form)) return false;
        var t = Preset(speciesId, shape, form);
        ulong h = IdHash(speciesId);
        double r(uint salt) => ReprId.Unit(h, salt);
        double v(uint salt, double spread) => 1.0 + (r(salt) * 2 - 1) * spread;

        var leafAd = PlantShapes.Srgb(colour[0], colour[1], colour[2]);
        var accent = PlantShapes.Srgb(colour2[0], colour2[1], colour2[2]);
        bool accentFlower = shape is "thornmantle" or "widowspire" or "heddlelace" or "bannerreed" or "tussock" or "ashwire";
        bool accentFruit = shape is "witchberry";
        // color2 is the lighter foliage (underside) for most species, but the flower/fruit/straw accent for a few
        var leafAb = accentFlower || accentFruit ? FormShapes.Lerp(leafAd, PlantShapes.Srgb(0.62, 0.78, 0.5), 0.35) : accent;
        bool glowing = glow != null && (glow[0] + glow[1] + glow[2]) > 0.05;
        t = t with
        {
            HeightM = Math.Max(0.03, heightM),
            CrownRadiusM = Math.Max(0.03, radiusM),
            LeafAdaxial = leafAd,
            LeafAbaxial = leafAb,
            LeafYoung = FormShapes.Lerp(leafAb, PlantShapes.Srgb(0.62, 0.78, 0.38), 0.4),
            BranchAngleDeg = t.BranchAngleDeg * v(1, 0.12),
            InternodeM = t.InternodeM * v(2, 0.15),
            LeafLengthM = t.LeafLengthM * v(3, 0.10),
            LeafAspect = Math.Clamp(t.LeafAspect * v(4, 0.10), 0.03, 1.1),
            Sag = t.Sag * v(5, 0.15),
            WanderDeg = t.WanderDeg * v(6, 0.2),
            LeafElevDeg = t.LeafElevDeg + (r(7) * 2 - 1) * 6,
            LeafCurl = t.LeafCurl * v(8, 0.15),
            ColourVariance = 0.04 + 0.06 * r(9),
            Emissive = t.Emissive || glowing && t.Inflorescence != InflorescenceType.None,
        };
        if (accentFlower) t = t with { FlowerColour = accent, FlowerColour2 = accent * 0.7 };
        if (accentFruit) t = t with { FruitColour = accent, FlowerColour = accent };
        if (glowing && t.Inflorescence != InflorescenceType.None)
        {
            var g = PlantShapes.Srgb(glow![0], glow[1], glow[2]);
            t = t with { FlowerColour = g, FlowerColour2 = leafAb * 0.9 };
        }
        if (t.Phyllotaxy is Phyllotaxy.Alternate137 or Phyllotaxy.Distichous && r(10) < 0.25 && form != GrowthForm.FernRosette)
            t = t with { Phyllotaxy = t.Phyllotaxy == Phyllotaxy.Alternate137 ? Phyllotaxy.Distichous : Phyllotaxy.Alternate137 };
        if (form == GrowthForm.MultiStemShrub || form == GrowthForm.Forb)
            t = t with { StemMax = Math.Max(t.StemMin, t.StemMax + (r(11) < 0.5 ? 0 : 1)) };
        traits = t;
        return true;
    }

    public static IPlantGenerator CreateGenerator(string speciesId, FormTraits traits) => traits.Form switch
    {
        GrowthForm.MultiStemShrub => new MultiStemShrubGenerator(speciesId, traits),
        GrowthForm.SingleTrunkTree => new SingleTrunkTreeGenerator(speciesId, traits),
        GrowthForm.FernRosette => new FernRosetteGenerator(speciesId, traits),
        GrowthForm.Forb => new ForbGenerator(speciesId, traits),
        GrowthForm.Graminoid => new GraminoidGenerator(speciesId, traits),
        _ => throw new NotSupportedException($"growth form {traits.Form} is not implemented"),
    };

    // ------------------------------------------------------------------ presets

    private static Vec3 S(double r, double g, double b) => PlantShapes.Srgb(r, g, b);

    private static FormTraits Preset(string id, string shape, GrowthForm form) => form switch
    {
        GrowthForm.MultiStemShrub => Shrub(id, shape),
        GrowthForm.SingleTrunkTree => Tree(id, shape),
        GrowthForm.FernRosette => Fern(id, shape),
        GrowthForm.Forb => Forb(id, shape),
        _ => Grass(id, shape),
    };
